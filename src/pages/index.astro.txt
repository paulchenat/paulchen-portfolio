  <script is:inline>
    const nav = document.getElementById('mainNav');
    window.addEventListener('scroll', () => {
      if (window.scrollY > (window.innerHeight * 0.35)) {
        nav.classList.add('collapsed');
      } else {
        nav.classList.remove('collapsed');
      }
    });

    document.querySelectorAll('.scrub-container').forEach(box => {
      const items = box.querySelectorAll('.scrub-item');
      const count = items.length;
      if (count > 1) {
        box.addEventListener('mousemove', (e) => {
          if (window.innerWidth <= 768) return;
          const rect = box.getBoundingClientRect();
          const activeIndex = Math.floor(Math.max(0, Math.min(0.999, (e.clientX - rect.left) / rect.width)) * count);
          items.forEach((item, i) => item.classList.toggle('active', i === activeIndex));
        });
        box.addEventListener('mouseleave', () => {
          items.forEach((item, i) => item.classList.toggle('active', i === 0));
        });
      }
    });

    document.querySelectorAll('.arch-info-btn').forEach(btn => {
      btn.addEventListener('click', () => {
        if (window.innerWidth > 768) return;
        const row = btn.closest('.archive-project-row');
        row.querySelector('.mobile-info-drawer').classList.toggle('open');
      });
    });

    const lb = document.getElementById('globalLightbox');
    document.querySelectorAll('.archive-item-1-1').forEach(item => {
      item.addEventListener('click', () => {
        document.getElementById('lbTitle').innerText = item.dataset.title + ' (' + item.dataset.year + ')';
        document.getElementById('lbCopyright').innerText = '© ' + item.dataset.copyright;
        lb.classList.add('active');
        document.body.style.overflow = 'hidden';
      });
    });
    document.getElementById('lbClose')?.addEventListener('click', () => {
      lb.classList.remove('active');
      document.body.style.overflow = 'auto';
    });
  </script>
